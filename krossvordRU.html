import tkinter as tk # Слова для кроссворда words = { "елка": {"dir": "v", "row": 0, "col": 2}, "снег": {"dir": "h", "row": 3, "col": 1}, "метель": {"dir": "v", "row": 3, "col": 4}, "мандарины": {"dir": "h", "row": 13, "col": 0}, "каток": {"dir": "h", "row": 2, "col": 0}, "хоккей": {"dir": "v", "row": 2, "col": 9}, "лед": {"dir": "h", "row": 6, "col": 4}, "гололед": {"dir": "h", "row": 8, "col": 0}, "зима": {"dir": "v", "row": 0, "col": 7}, "салаты": {"dir": "v", "row": 7, "col": 12}, } # Создадим сетку 15x15, заполненную пробелами GRID_SIZE = 15 grid = [["" for _ in range(GRID_SIZE)] for _ in range(GRID_SIZE)] # Заполним сетку буквами for word, info in words.items(): r, c = info["row"], info["col"] if info["dir"] == "h": # горизонтально for i, ch in enumerate(word): grid[r][c + i] = ch.upper() elif info["dir"] == "v": # вертикально for i, ch in enumerate(word): grid[r + i][c] = ch.upper() # Окно Tkinter root = tk.Tk() root.title("Новогодний кроссворд") root.configure(bg="#ffe4b5") # цвет фона — «мокко» # Размеры ячейки CELL_SIZE = 40 canvas = tk.Canvas(root, width=GRID_SIZE * CELL_SIZE, height=GRID_SIZE * CELL_SIZE + 200, bg="#ffe4b5") canvas.pack() # Рисуем сетку и буквы for r in range(GRID_SIZE): for c in range(GRID_SIZE): x1 = c * CELL_SIZE y1 = r * CELL_SIZE x2 = x1 + CELL_SIZE y2 = y1 + CELL_SIZE # Заливка: белая — буква, серая — пусто fill_color = "white" if grid[r][c] else "#d3d3d3" canvas.create_rectangle(x1, y1, x2, y2, fill=fill_color, outline="black") if grid[r][c]: canvas.create_text( x1 + CELL_SIZE // 2, y1 + CELL_SIZE // 2, text=grid[r][c], font=("Arial", 16, "bold"), fill="black" ) else: # Нумерация только для клеток с началом слов (упрощённо — по первым буквам, которые не в центре слова) pass # можно добавить нумерацию при желании # Подписи: загадки/вопросы clues = [ "1. Весёлая зелёная красавица в доме под Новый год", "2. Белый ковёр, покрывающий землю зимой", "3. Сильный ветер со снегом", "4. Яркие ароматные фрукты — символ Нового года", "5. Место, где катаются на коньках", "6. Командная зимняя игра с шайбой", "7. Твёрдая вода", "8. Очень скользкая дорога после дождя и мороза", "9. Холодное время года с праздниками", "10. Оливье и «селёдка под шубой» — это…", ] clue_frame = tk.Frame(root, bg="#ffe4b5") clue_frame.pack(pady=10) tk.Label(clue_frame, text="Вопросы:", font=("Arial", 14, "bold"), bg="#ffe4b5").pack(anchor="w") for i, clue in enumerate(clues, 1): tk.Label(clue_frame, text=f"{i}. {clue}", bg="#ffe4b5", font=("Arial", 10), anchor="w", justify="left").pack(anchor="w") # Ответы (скрытые, но можно раскомментировать или показать по кнопке) answers = ["елка", "снег", "метель", "мандарины", "каток", "хоккей", "лед", "гололед", "зима", "салаты"] def show_answers(): ans_win = tk.Toplevel(root) ans_win.title("Ответы") ans_win.configure(bg="white") tk.Label(ans_win, text="Ответы:", font=("Arial", 14, "bold"), bg="white").pack(pady=5) for i, ans in enumerate(answers, 1): tk.Label(ans_win, text=f"{i}. {ans}", bg="white", font=("Arial", 12)).pack(anchor="w") btn = tk.Button(root, text="Показать ответы", command=show_answers, bg="#4CAF50", fg="white", font=("Arial", 10)) btn.pack(pady=10) root.mainloop()
